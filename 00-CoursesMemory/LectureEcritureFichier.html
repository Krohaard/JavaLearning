Lecture et Écriture de Fichiers en Java

Chapitre 1 : Introduction aux Flux en Java

Théorie : En Java, les opérations de lecture et d'écriture de fichiers sont gérées par des flux (streams). Les principaux types de flux sont :

  • InputStream et OutputStream pour les données binaires.
  • Reader et Writer pour les données textuelles.
Les classes concrètes comme FileInputStream, FileOutputStream, FileReader, et FileWriter sont utilisées pour manipuler les fichiers.

Exemple 1 : Écriture dans un fichier texte

import java.io.FileWriter;
import java.io.IOException;

public class EcritureFichier {
    public static void main(String[] args) {
        try (FileWriter writer = new FileWriter("exemple.txt")) {
            writer.write("Bonjour, ceci est un test d'écriture dans un fichier.");
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}

Exemple 2 : Lecture d'un fichier texte

import java.io.FileReader;
import java.io.IOException;

public class LectureFichier {
    public static void main(String[] args) {
        try (FileReader reader = new FileReader("exemple.txt")) {
            int character;
            while ((character = reader.read()) != -1) {
                System.out.print((char) character);
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}

Exemple 3 : Écriture avec Buffer

import java.io.BufferedWriter;
import java.io.FileWriter;
import java.io.IOException;

public class EcritureBuffer {
    public static void main(String[] args) {
        try (BufferedWriter writer = new BufferedWriter(new FileWriter("exemple_buffer.txt"))) {
            writer.write("Utilisation d'un buffer pour écrire dans un fichier.");
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}

Chapitre 2 : Gestion des Exceptions

Théorie : Lors de la manipulation de fichiers, il est crucial de gérer les exceptions pour éviter les erreurs d'exécution. Les exceptions courantes incluent IOException et FileNotFoundException. Utiliser des blocs try-with-resources permet de fermer automatiquement les flux après utilisation.

Exemple 1 : Gestion des exceptions avec try-with-resources

import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;

public class LectureAvecExceptions {
    public static void main(String[] args) {
        try (BufferedReader reader = new BufferedReader(new FileReader("exemple.txt"))) {
            String line;
            while ((line = reader.readLine()) != null) {
                System.out.println(line);
            }
        } catch (IOException e) {
            System.err.println("Erreur de lecture du fichier : " + e.getMessage());
        }
    }
}

Exemple 2 : Écriture avec gestion des exceptions

import java.io.BufferedWriter;
import java.io.FileWriter;
import java.io.IOException;

public class EcritureAvecExceptions {
    public static void main(String[] args) {
        try (BufferedWriter writer = new BufferedWriter(new FileWriter("exemple_exceptions.txt"))) {
            writer.write("Gestion des exceptions lors de l'écriture.");
        } catch (IOException e) {
            System.err.println("Erreur d'écriture dans le fichier : " + e.getMessage());
        }
    }
}

Exemple 3 : Lecture et écriture combinées

import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.FileReader;
import java.io.FileWriter;
import java.io.IOException;

public class LectureEcritureCombinee {
    public static void main(String[] args) {
        String sourceFile = "exemple.txt";
        String destinationFile = "copie_exemple.txt";

        try (BufferedReader reader = new BufferedReader(new FileReader(sourceFile));
             BufferedWriter writer = new BufferedWriter(new FileWriter(destinationFile))) {

            String line;
            while ((line = reader.readLine()) != null) {
                writer.write(line);
                writer.newLine();
            }
        } catch (IOException e) {
            System.err.println("Erreur lors de la copie du fichier : " + e.getMessage());
        }
    }
}
